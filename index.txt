<script>
    const { Renderer, Stave, StaveNote, Accidental, Barline } = Vex.Flow;
    const inputElement = document.getElementById('music-input');
    const errorElement = document.getElementById('error-log');
    const outputDiv = document.getElementById('boo');

    function parseMusicText(text) {
        const tokens = text.trim().split(/\s+/).filter(t => t.length > 0);
        const parsedItems = [];
        
        // Regex per le note (accetta da 1 a 4 segni + o -)
        const noteRegex = /^([db])?([A-G])(\d+)([\+\-]{1,4})?$/;

        for (let i = 0; i < tokens.length; i++) {
            const token = tokens[i];

            // Se incontra il simbolo di battuta
            if (token === '|') {
                parsedItems.push({ type: 'barline' });
                continue;
            }

            // Altrimenti controlla se è una nota valida
            const match = token.match(noteRegex);
            if (!match) throw new Error(`Elemento non valido rilevato: "${token}"`);

            const alt = match[1] || "";       
            const noteLetter = match[2];     
            const duration = match[3];       
            const octaveSigns = match[4] || ""; 

            let octave = 4; 
            if (octaveSigns.includes('+')) octave += octaveSigns.length;
            if (octaveSigns.includes('-')) octave -= octaveSigns.length;

            const key = `${noteLetter.toLowerCase()}/${octave}`;
            let accidentalType = null;
            if (alt === 'd') accidentalType = '#';
            if (alt === 'b') accidentalType = 'b';

            parsedItems.push({
                type: 'note',
                keys: [key],
                duration: duration,
                accidental: accidentalType
            });
        }
        return parsedItems;
    }

    function renderScore() {
        outputDiv.innerHTML = "";
        errorElement.innerText = "";
        const inputText = inputElement.value;
        if (!inputText.trim()) return;

        try {
            const itemsData = parseMusicText(inputText);
            if (itemsData.length === 0) return;

            // Calcola la larghezza in base agli elementi (le barlines occupano meno spazio delle note)
            let estimatedWidth = 100;
            itemsData.forEach(item => {
                estimatedWidth += (item.type === 'barline') ? 30 : 80;
            });
            const width = Math.max(450, estimatedWidth);

            const renderer = new Renderer(outputDiv, Renderer.Backends.SVG);
            renderer.resize(width, 200);
            const context = renderer.getContext();

            const stave = new Stave(10, 40, width - 20);
            stave.addClef("treble").setContext(context).draw();

            const staveNotes = [];
            itemsData.forEach((data, index) => {
                if (data.type === 'barline') {
                    // Crea una linea di battuta (non l'ultima, una normale stanghetta)
                    // In VexFlow aggiungiamo un elemento speciale di battuta se non è alla fine del pentagramma stesso
                    const barNote = new StaveNote({ keys: ["b/4"], duration: "b", type: "b" });
                    // Nota: Per semplicità visiva in questa configurazione lineare usiamo le barlines native di formattazione
                    // o una nota invisibile modificata. Usiamo il modificatore nativo di Vexflow per la formattazione corretta.
                } else {
                    const note = new StaveNote({ keys: data.keys, duration: data.duration });
                    if (data.accidental) note.addModifier(new Accidental(data.accidental));
                    staveNotes.push(note);
                }
            });

            // Per gestire al meglio l'inserimento visivo delle linee di battuta nel flusso di note, 
            // ricreiamo l'array combinato accettato dal Formatter di VexFlow v4
            const finalTickables = [];
            itemsData.forEach(data => {
                if (data.type === 'barline') {
                    // Inserisce una stanghetta nel flusso
                    const bar = new Vex.Flow.BarNote();
                    finalTickables.push(bar);
                } else {
                    const note = new StaveNote({ keys: data.keys, duration: data.duration });
                    if (data.accidental) note.addModifier(new Accidental(data.accidental));
                    finalTickables.push(note);
                }
            });

            Vex.Flow.Formatter.FormatAndDraw(context, stave, finalTickables);
        } catch (error) {
            errorElement.innerText = error.message;
        }
    }

    inputElement.addEventListener('input', renderScore);
    renderScore();
</script>
